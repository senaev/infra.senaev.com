### Fix: /etc/network/interfaces on the host (only the post-up line is new)
auto lo
iface lo inet loopback

iface nic0 inet manual
	post-up /usr/sbin/ethtool -K nic0 tso off gso off gro off

auto vmbr0
iface vmbr0 inet dhcp
	bridge-ports nic0
	bridge-stp off
	bridge-fd 0

source /etc/network/interfaces.d/*

$ ifup --syntax-check -a
warning: vmbr0: bridge-fd: value of out range "0": valid attribute range: 2-255
(The warning is old: the Proxmox installer writes bridge-fd 0. It is not about the new line.)
